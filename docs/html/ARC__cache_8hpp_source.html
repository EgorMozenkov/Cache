122template <typename Key>
123CacheResult<Key> ARC_Cache<Key>::request(const Key& key)
124{
125 auto it = ARC_map.find(key);
126 CacheResult<Key> result = {false, false, Key{}};
127
128 // Ключ найден в хэш таблицe
129 if (it != ARC_map.end()) {
130
131 // Нашли в MRU
132 if (it->second.type == QueueType::MRU) {
133
134 Log::trace(Log::DEBUG, "ХИТ: Ключ ", key, " в MRU, переносим его в начало MFU\n");
135
136 MRU_list.erase (it->second.list_it);
137 MFU_list.push_front(key);
138
139 it->second.type = QueueType::MFU;
140 it->second.list_it = MFU_list.begin();
141
142 result.hit = true;
143 return result;
144 }
145
146 // Нашли в MFU
147 if (it->second.type == QueueType::MFU) {
148
149 Log::trace(Log::DEBUG, "ХИТ: Ключ ", key, " в MFU, переносим его в начало MFU\n");
150
151 MFU_list.splice(MFU_list.begin(), MFU_list, it->second.list_it);
152
153 result.hit = true;
154 return result;
155 }
156
157 // Нашли в MFU_ghost
158 if (it->second.type == QueueType::MFU_ghost) {
159
160 Log::trace(Log::DEBUG, "МИСС: Ключ ", key, " ключ найден в MFU_ghost\n");
161
162 size_t delta = 0;
163 if (MFU_ghost_list.size() >= MRU_ghost_list.size()) {
164 delta = 1;
165 }
166 else {
167 delta = MRU_ghost_list.size() / MFU_ghost_list.size();
168 }
169
170 if (p > delta) {
171 p = std::min(capacity, p - delta);
172 }
173 else {
174 p = 0;
175 }
176 replace(key, result);
177
178 MFU_ghost_list.erase (it->second.list_it);
179 MFU_list.push_front(key);
180
181 it->second.type = QueueType::MFU;
182 it->second.list_it = MFU_list.begin();
183
184 return result;
185 }
186
187 //Нашли в MRU_ghost
188 if (it->second.type == QueueType::MRU_ghost) {
189
190 Log::trace(Log::DEBUG, "МИСС: Ключ ", key, " ключ найден в MRU_ghost\n");
191
192 size_t delta = 0;
193 if (MRU_ghost_list.size() >= MFU_ghost_list.size()) {
194 delta = 1;
195 }
196 else {
197 delta = MFU_ghost_list.size() / MRU_ghost_list.size();
198 }
199
200
201 p = std::min(capacity, p + delta);
202 replace(key, result);
203
204 MRU_ghost_list.erase (it->second.list_it);
205 MFU_list.push_front(key);
206 ARC_map[key] = {QueueType::MFU, MFU_list.begin()};
207
208 return result;
209 }
210
211 }
212
213 // Ключа нет в хэш-таблице
214
215 if (MRU_list.size() + MRU_ghost_list.size() == capacity) {
216 if (MRU_list.size() < capacity) {
217 // Удаляем самого старого из MRU_ghost
218 ARC_map.erase(MRU_ghost_list.back());
219 MRU_ghost_list.pop_back();
220
221 }
222 else {
223 // в MRU_ghost его нет
224
225 result.has_evicted = true;
226 result.evicted_key = MRU_list.back();
227
228 ARC_map.erase(MRU_list.back());
229 MRU_list.pop_back();
230 }
231 }
232 else if (MRU_list.size() + MFU_list.size() + MRU_ghost_list.size() + MFU_ghost_list.size() == 2 * capacity) {
233 ARC_map.erase(MFU_ghost_list.back());
234 MFU_ghost_list.pop_back();
235 }
236
237
238 if (MRU_list.size() + MFU_list.size() >= capacity) {
239 replace(key, result);
240 }
241
242 Log::trace(Log::TRACE, "МИСС: Ключ ", key, " не найден и добавлен в MRU\n");
243 MRU_list.push_front(key);
244 ARC_map[key] = {QueueType::MRU, MRU_list.begin()};
245
246 return result;
247}

267{

269
270 if (it_map != ARC_map.end()) {

272 if(it_map->second.type == QueueType::MFU) {

279 MFU_ghost_list.erase(it_map->second.list_it);

282 MRU_ghost_list.erase(it_map->second.list_it);
