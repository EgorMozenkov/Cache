72template <typename Key>
73CacheResult<Key> TwoQ_Cache<Key>::request(const Key& key)
74{
75 auto it = TwoQ_map.find(key);
76 CacheResult<Key> result = {false, false, Key{}};
77
78 // Элемента нет ни в одной из очередей
79 if (it == TwoQ_map.end()) {
80
81 // Проверяем переполнение IN
82 if (IN_list.size() >= size_IN) {
83 Key value_1 = IN_list.back();
84 IN_list.pop_back();
85
86 // Проверяем переполнение OUT
87 if (OUT_list.size() >= size_OUT) {
88 Key value_2 = OUT_list.back();
89 Log::trace(Log::DEBUG, "Вытеснение: Удаляем старейший ключ из OUT: ", value_2, "\n");
90 OUT_list.pop_back();
91 TwoQ_map.erase(value_2);
92
93 result.has_evicted = true;
94 result.evicted_key = value_2;
95 }
96
97 // Переносим элемент из IN в OUT
98 OUT_list.push_front(value_1);
99 TwoQ_map[value_1] = {QueueType::OUT, OUT_list.begin()};
100 }
101
102 // Добавляем новый ключ в IN
103 IN_list.push_front(key);
104 TwoQ_map[key] = {QueueType::IN, IN_list.begin()};
105
106 Log::trace(Log::DEBUG, "МИСС: Ключ ", key, " добавлен в IN\n");
107 return result;
108 }
109
110 // Ключ найден
111 auto& node = it->second;
112
113 if (node.type == QueueType::VIP) {
114 VIP_list.splice(VIP_list.begin(), VIP_list, node.list_it);
115 Log::trace(Log::DEBUG, "ХИТ: Ключ ", key, " в VIP, переносим в начало\n");
116
117 result.hit = true;
118 return result;
119 }
120 else if (node.type == QueueType::IN) {
121 Log::trace(Log::DEBUG, "ХИТ: Ключ ", key, " в IN, оставляем на месте\n");
122
123 result.hit = true;
124 return result;
125 }
126 else if (node.type == QueueType::OUT) {
127
128 // Проверяем переполнение VIP
129 if (VIP_list.size() >= size_VIP) {
130 Key evicted_vip = VIP_list.back();
131 Log::trace(Log::DEBUG, "Вытеснение: VIP переполнен. Удаляем: ", evicted_vip, "\n");
132 TwoQ_map.erase(evicted_vip);
133 VIP_list.pop_back();
134
135 result.has_evicted = true;
136 result.evicted_key = evicted_vip;
137 }
138
139 OUT_list.erase(node.list_it);
140
141 VIP_list.push_front(key);
142
143 node.type = QueueType::VIP;
144 node.list_it = VIP_list.begin();
145
146 Log::trace(Log::DEBUG, "МИСС: Ключ ", key, " повышен из OUT в VIP\n");
147 return result;
148 }
149
150 return result;
151}

169template <typename Key>
170void TwoQ_Cache<Key>::erase(const Key& key)
171{
172 auto it_map = TwoQ_map.find(key);
173
174 if (it_map != TwoQ_map.end()) {
175
176 if(it_map->second.type == QueueType::IN) {
177 IN_list.erase(it_map->second.list_it);
178 }
179 else if(it_map->second.type == QueueType::VIP) {
180 VIP_list.erase(it_map->second.list_it);
181 }
182 else if(it_map->second.type == QueueType::OUT) {
183 OUT_list.erase(it_map->second.list_it);
184 }
185
186 TwoQ_map.erase(it_map);
187 }
