58template <typename Key>
59CacheResult<Key> LFU_Cache<Key>::request(const Key& key)
60{
61
62 if (capacity == 0) return {false, false, Key{}};
63
64 CacheResult<Key> result = {false, false, Key{}};
65
66 Log::trace(Log::DEBUG, "Запрошен ключ: ", key, "\n");
67 auto it = LFU_map_key.find(key);
68
69
70 if (it != LFU_map_key.end()) {
71
72 Log::trace(Log::DEBUG, "ХИТ: Ключ ", key, " найден в кэше, переносим в начало\n");
73 int freq_before = LFU_map_key[key].freq;
74 LFU_map_key[key].freq++;
75
76 // Перемещаем элемент в список с freq + 1
77 LFU_map_freq[freq_before + 1].splice
78 (LFU_map_freq[freq_before + 1].begin(),
79 LFU_map_freq[freq_before],
80 LFU_map_key[key].list_it);
81
82 // Проверяем не остался ли старый список пустым
83 if (LFU_map_freq[freq_before].empty() && min_freq == freq_before) {
84 min_freq++;
85 }
86
87 result.hit = true;
88 result.has_evicted = false;
89 result.evicted_key = Key{};
90 return result;
91 }
92
93 // Проверяем переполнение
94 if (LFU_map_key.size() == capacity) {
95
96 Key value_delete = LFU_map_freq[min_freq].back();
97 Log::trace(Log::DEBUG, "Вытеснение: Кэш переполнен. Удаляем самый не популярный ключ: ", value_delete, "\n");
98
99 LFU_map_freq[min_freq].pop_back();
100 LFU_map_key.erase(value_delete);
101
102 result.has_evicted = true;
103 result.evicted_key = value_delete;
104 }
105
106 // Ставим в кэш новое значение
107 min_freq = 1;
108 LFU_map_freq[min_freq].push_front(key);
109 LFU_map_key[key].freq = min_freq;
110 LFU_map_key[key].list_it = LFU_map_freq[min_freq].begin();
111
112 Log::trace(Log::DEBUG, "МИСС: Ключ ", key, " добавлен в кэш\n");
113
114 result.hit = false;
115 return result;
116}
